  --from-literal=github_token="$PAT" --dry-run=client -o yaml \
| kubeseal --format=yaml --controller-name=sealed-secrets \
  --controller-namespace=kube-system \
  > secrets/kanban-runner-pat.sealed.yaml

  --controller-namespace=kube-system \
  > secrets/argocd-image-updater-git.sealed.yaml
